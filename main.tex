\newcommand{\pageSectionAnnonce}[1]{%
	\clearpage
	\thispagestyle{styleSansPagination}
	\vspace*{\fill}
	\begin{center}
		\begin{tikzpicture}
			\node[
			fill=vertCongo,
			text=blancPur,
			font=\Huge\bfseries,
			minimum width=14cm,
			minimum height=3.5cm,
			rounded corners=8pt,
			inner sep=16pt,
			align=center
			] {#1};
		\end{tikzpicture}
	\end{center}
	\vspace*{\fill}
	\clearpage
}

	\thispagestyle{styleSansPagination}
	\vspace*{\fill}
	\begin{center}
		\begin{tikzpicture}
			\node[
			fill=vertCongo,
			text=blancPur,
			font=\fontsize{16}{20}\selectfont\bfseries,
			minimum width=13cm,
			minimum height=2.8cm,
			rounded corners=6pt,
			inner sep=14pt,
			align=center
			] {#1\\[4pt]\normalsize\mdseries #2};
		\end{tikzpicture}
	\end{center}
	\vspace*{\fill}
	\clearpage
}
