\vspace*{2cm}

\begin{flushright}
	\itshape
	
	\textit{À mon oncle,}\\[0.3cm]
	\textit{pour tout ce qu'il représente à mes yeux.}\\[0.3cm]
	\textit{Ce travail est pour toi.}
	
\end{flushright}

\vspace*{2cm}
\noindent\rule{\textwidth}{0.4pt}
\vspace*{1.5cm}

\begin{flushright}
	\itshape
	
	\textit{À mes parents,}\\[0.3cm]
	\textit{pour leur soutien indéfectible et les sacrifices consentis}\\[0.3cm]
	\textit{tout au long de mon parcours.}\\[0.6cm]
	
	\textit{À mon frère,}\\[0.3cm]
	\textit{pour son soutien matériel sans lequel ce travail}\\[0.3cm]
	\textit{n'aurait pas été possible.}\\[0.6cm]
	
	\textit{À tous ceux qui ont prié pour ma réussite,}\\[0.3cm]
	\textit{ces prières ont porté leurs fruits.}
	
\end{flushright}
